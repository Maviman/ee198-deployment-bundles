# The exact Python package versions the offline Orin packages carry.
# Pinned to what has actually run here, not to "latest": the packages are built
# on a PC with internet and installed on Orins without it, so a surprise there
# cannot be fixed with a quick `pip install`.
#
# On the Orin the installer only adds what is MISSING; a package that already
# works is never upgraded or downgraded. These pins decide what a fresh install gets.

# The pip wheel on the first Orin (HANDOFF.md). opencv-python 5.x now exists and
# is the newest, but it is untested on this hardware.
opencv-python==4.10.0.84

# 1.27.0 is what the first Orin and the dev PC run. 1.23.2 is the newest release
# that still supports Python 3.10 (Ubuntu 22.04 / JetPack 6).
onnxruntime==1.27.0 ; python_version >= "3.11"
onnxruntime==1.23.2 ; python_version < "3.11"

# The dev PC's selftest versions. numpy 2.2.6 is the newest for Python 3.10.
gymnasium==1.3.0
numpy==2.4.6 ; python_version >= "3.11"
numpy==2.2.6 ; python_version < "3.11"
pyyaml==6.0.3
